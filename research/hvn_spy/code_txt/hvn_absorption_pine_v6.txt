// Volume-by-price of the n bars starting `off` bars back. Each bar's volume is
// spread over the rows its high-low range overlaps, then Gaussian-smoothed.
buildProfile(int off, int n, int nRows, float sig) =>
    int last = hiHist.size() - 1
    float pHi = hiHist.get(last - off)
    float pLo = loHist.get(last - off)
    for i = off to off + n - 1
        pHi := math.max(pHi, hiHist.get(last - i))
        pLo := math.min(pLo, loHist.get(last - i))
    float step = (pHi - pLo) / nRows
    array<float> raw = array.new<float>(nRows, 0.0)
    array<float> rbuy = array.new<float>(nRows, 0.0)
    if step > 0
        for i = off to off + n - 1
            float h = hiHist.get(last - i)
            float l = loHist.get(last - i)
            float v = volHist.get(last - i)
            float bshare = shareHist.get(last - i)
            float r = h - l
            if r <= 0
                int k = math.max(0, math.min(nRows - 1, int((h - pLo) / step)))
                raw.set(k, raw.get(k) + v)
                rbuy.set(k, rbuy.get(k) + v * bshare)
            else
                int k0 = math.max(0, math.min(nRows - 1, int((l - pLo) / step)))
                int k1 = math.max(k0, math.min(nRows - 1, int((h - pLo) / step)))
                for k = k0 to k1
                    float bLo = pLo + k * step
                    float ov = math.min(h, bLo + step) - math.max(l, bLo)
                    if ov > 0
                        raw.set(k, raw.get(k) + v * ov / r)
                        rbuy.set(k, rbuy.get(k) + v * bshare * ov / r)
    array<float> sm = array.new<float>(nRows, 0.0)
    int rad = sig > 0 ? math.max(1, int(math.round(3 * sig))) : 0
    for k = 0 to nRows - 1
        if rad == 0
            sm.set(k, raw.get(k))
        else
            float acc = 0.0
            float wsum = 0.0
            for j = -rad to rad
                float w = math.exp(-0.5 * math.pow(j / sig, 2))
                wsum += w
                int kk = k + j
                if kk >= 0 and kk < nRows
                    acc += w * raw.get(kk)
            sm.set(k, acc / wsum)
    [pLo, step, sm, raw, rbuy]

// HVN detection: local peaks above the percentile threshold, biggest first,
// at least `minSep` rows apart. Returns [peak rows, zone low rows, zone high rows].
nodesOf(array<float> sm, int maxN) =>
    int n = sm.size()
    float cutoff = sm.percentile_linear_interpolation(thrPct)
    float pocV = sm.max()
    array<int> peaks = array.new<int>()
    if pocV > 0
        for k = 0 to n - 1
            float v = sm.get(k)
            float left = k > 0 ? sm.get(k - 1) : -1.0
            float right = k < n - 1 ? sm.get(k + 1) : -1.0
            if v >= left and v > right and v >= cutoff
                peaks.push(k)
    array<int> chosen = array.new<int>()
    array<int> zl = array.new<int>()
    array<int> zh = array.new<int>()
    while peaks.size() > 0 and chosen.size() < maxN
        int best = 0
        for p = 0 to peaks.size() - 1
            if sm.get(peaks.get(p)) > sm.get(peaks.get(best))
                best := p
        int k = peaks.remove(best)
        bool farEnough = true
        for c in chosen
            if math.abs(k - c) < minSep
                farEnough := false
        if farEnough
            float pk = sm.get(k)
            int lo = k
            int hi = k
            while lo > 0 and sm.get(lo - 1) >= pk * zoneRatio / 100 and sm.get(lo - 1) <= sm.get(lo)
                lo -= 1
            while hi < n - 1 and sm.get(hi + 1) >= pk * zoneRatio / 100 and sm.get(hi + 1) <= sm.get(hi)
                hi += 1
            chosen.push(k)
            zl.push(lo)
            zh.push(hi)
    [chosen, zl, zh]

    bool inside = low <= nd.hi and high >= nd.lo
    bool live = nd.active or nd.period == pId
    if inside and live
        atNode := true
    if live and lvlHeavyVol
        // bearish: pushed up into the node, rejected back below it; buyers absorbed
        if open < nd.lo and high >= nd.lo and close < nd.lo and upWick >= lvlWickE and (not hasDelta or dRatioL >= minDelta)
            lvlDir := -1
            lvlPrice := nd.price
            lvlZLo := nd.fullLo
            lvlZHi := nd.fullHi

        if open > nd.hi and low <= nd.hi and close > nd.hi and dnWick >= lvlWickE and (not hasDelta or dRatioL <= -minDelta)
            lvlDir := 1
            lvlPrice := nd.price
            lvlZLo := nd.fullLo
            lvlZHi := nd.fullHi

        if inside and nd.armed
            nd.active := false
            hvnRetest := true
            pocRetest := pocRetest or nd.isPoc

            nd.bx.set_right(bar_index)
            nd.lb.set_x(bar_index)
        else

                nd.armed := true
            if extendLvls
                nd.bx.set_right(bar_index + 1)
                nd.lb.set_x(bar_index + 1)

// Same rules as the research simulation: entry at the confirmation close + slippage, stop first if
// stop and target are hit in the same bar, gaps fill at the open, target as limit (no slippage),
// open trades closed at the close of the bar that ends at/after 16:00 New York. Designed for RTH charts.
type Trade
    int   dir
    float entry
    float stop
    float tgt
    float risk
    int   bar
var array<Trade> openTr = array.new<Trade>()
var int trN = 0
var int trWin = 0
var float trSumR = 0.0
var float trWinR = 0.0
var float trLossR = 0.0
var int trLossN = 0
var int trFastStop = 0
float tick = syminfo.mintick * slipTicks
bool eodBar = hour(time_close, "America/New_York") * 60 + minute(time_close, "America/New_York") >= 16 * 60
if openTr.size() > 0
    for k = openTr.size() - 1 to 0
        Trade tt = openTr.get(k)
        if bar_index > tt.bar
            float exitFill = na
            bool stopped = false
            bool gapStop = tt.dir == 1 ? open <= tt.stop : open >= tt.stop
            bool gapTgt = tt.dir == 1 ? open >= tt.tgt : open <= tt.tgt
            bool hitStop = tt.dir == 1 ? low <= tt.stop : high >= tt.stop
            bool hitTgt = tt.dir == 1 ? high >= tt.tgt : low <= tt.tgt
            if gapStop
                exitFill := open - tt.dir * tick
                stopped := true
            else if gapTgt
                exitFill := open
            else if hitStop
                exitFill := tt.stop - tt.dir * tick
                stopped := true
            else if hitTgt
                exitFill := tt.tgt
            else if eodBar
                exitFill := close
            if not na(exitFill)
                float rNet = ((exitFill - (tt.entry + tt.dir * tick)) * tt.dir - 2 * commShare) / tt.risk
                trN += 1
                trSumR += rNet
                if rNet > 0
                    trWin += 1
                    trWinR += rNet
                else
                    trLossN += 1
                    trLossR += rNet
                if stopped and bar_index - tt.bar <= 2
                    trFastStop += 1
                openTr.remove(k)
